5. CONFIGURACION DE NGINX
----------------------------------------------------
nano /etc/nginx/sites-available/easydrill

# Contenido del archivo:
server {
    listen 443 ssl;
    server_name easydrillsolutions.com www.easydrillsolutions.com;
    root /var/www/easydrill/browser;
    index index.html;
    ssl_certificate /etc/letsencrypt/live/easydrillsolutions.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/easydrillsolutions.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
    location / {
        try_files $uri $uri/ /index.html;
    }
    location /api {
        proxy_pass http://localhost:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}
server {
    listen 80;
    server_name easydrillsolutions.com www.easydrillsolutions.com;
    return 301 https://$host$request_uri;
}

ln -s /etc/nginx/sites-available/easydrill /etc/nginx/sites-enabled/
nginx -t
systemctl restart nginx
